## 文件一：utils/logger.py

这是**单例模式的日志工具**，防止日志重复初始化，同时支持控制台和文件双输出。

```python
import logging
import os
import sys
from typing import Optional

_logger_instance: Optional[logging.Logger] = None


def get_logger(name: str = "codemind") -> logging.Logger:
    """
    获取或创建单例日志实例。

    Args:
        name: 日志名称

    Returns:
        配置好的日志实例
    """
    global _logger_instance

    # 如果已存在，直接返回（单例核心）
    if _logger_instance is not None:
        return _logger_instance

    # 创建新的 logger，基础级别设为 DEBUG（确保所有级别的日志都能被接收）
    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)

    # 避免重复添加 handler
    if logger.handlers:
        _logger_instance = logger
        return logger

    # 日志格式
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # 控制台输出 handler（INFO 级别）
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(logging.INFO)
    console_handler.setFormatter(formatter)
    logger.addHandler(console_handler)

    # 文件输出 handler（DEBUG 级别）
    log_dir = os.path.join(os.path.dirname(os.path.dirname(__file__)), "logs")
    os.makedirs(log_dir, exist_ok=True)
    file_handler = logging.FileHandler(
        os.path.join(log_dir, "app.log"),
        encoding="utf-8"
    )
    file_handler.setLevel(logging.DEBUG)
    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)

    logger.propagate = False  # 防止传播到父 logger

    _logger_instance = logger
    return logger
```

**知识点**：
- **单例模式**：确保整个应用只有一个 logger 实例
- **_logger_instance**：用全局变量保存实例
- **双输出配置**：
  - 控制台：INFO 级别，简洁输出
  - 文件：DEBUG 级别，详细记录，保存在 `logs/app.log`
- **logger.propagate = False**：防止日志重复输出

## 文件二：utils/config.py

这是**配置管理工具**，统一管理 config.yml 和 .env。

```python
import os
from typing import Any, Dict
import yaml
from dotenv import load_dotenv


class Config:
    """配置管理器，从 config.yml 和环境变量加载配置。"""

    _config: Dict[str, Any] = {}
    _loaded: bool = False

    @classmethod
    def load(cls, config_path: str = "config.yml") -> None:
        """从 YAML 文件和环境变量加载配置。"""
        if cls._loaded:
            return

        # 加载环境变量
        load_dotenv()

        # 加载 YAML 配置
        if os.path.exists(config_path):
            with open(config_path, 'r', encoding='utf-8') as f:
                cls._config = yaml.safe_load(f) or {}

        cls._loaded = True

    @classmethod
    def get(cls, key: str, default: Any = None) -> Any:
        """
        通过点分隔的键获取配置值。

        Args:
            key: 配置键，例如 "chroma.persist_dir"
            default: 默认值

        Returns:
            配置值
        """
        if not cls._loaded:
            cls.load()

        parts = key.split('.')
        value = cls._config

        for part in parts:
            if isinstance(value, dict) and part in value:
                value = value[part]
            else:
                return default

        return value

    @classmethod
    def get_env(cls, key: str, default: str = "") -> str:
        """获取环境变量。"""
        if not cls._loaded:
            cls.load()
        return os.getenv(key, default)
```

**知识点**：
- **类级别的缓存**：`_config` 和 `_loaded` 是类变量，全局共享
- **点号分隔访问**：`Config.get("chroma.persist_dir")` 访问嵌套配置
- **延迟加载**：第一次调用 `get()` 时才加载配置

## 文件四：utils/summarizer.py

这是**总结层模块**，负责对向量检索返回的结果进行总结提炼。将总结相关代码独立成模块，让代码结构更清晰、更易维护。

```python
"""
总结层模块（Summarizer Layer）

负责对向量检索返回的结果进行总结提炼，
避免将过多原始文档直接传给主 LLM。
"""
from typing import List
from langchain_core.prompts import PromptTemplate
from langchain_community.chat_models import ChatTongyi

from utils.logger import get_logger
from prompts.prompt_manager import PromptManager, PromptScenario, PromptLanguage

logger = get_logger("summarizer")


def get_summarizer_prompt_template() -> PromptTemplate:
    """
    获取总结提示词模板。
    
    Returns:
        总结提示词模板
    """
    prompt_manager = PromptManager.get_instance()
    return prompt_manager.get_prompt(
        scenario=PromptScenario.SUMMARIZATION,
        language=PromptLanguage.ZH_CN
    )


def build_context(docs: List) -> str:
    """
    从检索到的文档构建上下文字符串。

    Args:
        docs: 检索到的文档列表

    Returns:
        格式化后的上下文字符串
    """
    context_parts = []
    for i, doc in enumerate(docs, 1):
        source = doc.metadata.get('source', 'unknown')
        content = doc.page_content
        context_parts.append(f"### 来源 {i}: {source}\n```\n{content}\n```")
    return "\n\n".join(context_parts)


def summarize_context(question: str, raw_context: str, summarizer_llm: ChatTongyi) -> str:
    """
    使用总结模型对原始上下文进行总结。

    Args:
        question: 用户原始问题
        raw_context: 由检索文档构建的原始上下文
        summarizer_llm: 总结模型实例

    Returns:
        总结后的精炼内容
    """
    prompt_template = get_summarizer_prompt_template()
    prompt = prompt_template.format(input=question, context=raw_context)
    response = summarizer_llm.invoke(prompt)
    summary = response.content if hasattr(response, 'content') else str(response)

    logger.debug("Context summarization completed")
    return summary


def create_summarizer(model: str = "qwen-max", temperature: float = 0.1) -> ChatTongyi:
    """
    创建总结模型实例。

    Args:
        model: 模型名称
        temperature: 温度参数

    Returns:
        ChatTongyi 实例
    """
    logger.info(f"Creating summarizer LLM: model={model}, temperature={temperature}")
    return ChatTongyi(
        model=model,
        temperature=temperature,
    )
```

**知识点**：
- ✅ **模块化设计**：将总结相关功能独立成模块，职责单一
- ✅ **可复用性**：`build_context()`、`summarize_context()`、`create_summarizer()` 都可以在其他地方复用
- ✅ **类型提示**：使用 `typing` 模块提供类型注解，代码更易读
- ✅ **文档字符串**：每个函数都有清晰的 docstring，说明参数和返回值
- ✅ **统一提示词管理**：使用 `PromptManager` 获取提示词模板，支持版本控制和多语言

### 核心：extract_python_classes_and_functions 函数（新增）

这是**优化后的 Python 代码切分函数**，优先保留完整类。

```python
def extract_python_classes_and_functions(content: str, max_class_length: int = 3000):
    """
    提取 Python 类和函数，尽可能保留完整的类。

    策略：
    1. 先识别所有类定义
    2. 对于类：如果长度 < max_class_length，完整保留
    3. 对于超长类：再按方法切分
    4. 提取独立函数
    """
    blocks = []
    lines = content.split('\n')

    # 匹配任意缩进级别的 class 和 def
    class_pattern = re.compile(r'^(\s*)class\s+\w+')
    def_pattern = re.compile(r'^(\s*)def\s+\w+')

    i = 0
    n = len(lines)

    while i < n:
        line = lines[i]

        # ──────────────────────────────────
        # 第一步：识别类
        # ──────────────────────────────────
        class_match = class_pattern.match(line)
        if class_match:
            class_indent = class_match.group(1)  # 获取类的缩进
            class_start = i
            i += 1

            # 找到类的结束位置（下一个非缩进行或文件尾）
            while i < n:
                curr_line = lines[i]
                if curr_line.strip() and not curr_line.startswith(class_indent + ' ') and not curr_line.startswith(class_indent + '\t'):
                    break
                i += 1

            class_content = '\n'.join(lines[class_start:i])

            # 判断类长度，决定是否完整保留
            if len(class_content) <= max_class_length:
                if class_content.strip():
                    blocks.append(class_content)
            else:
                # 超长类：按方法切分
                method_blocks = _split_class_by_methods(class_content, class_indent)
                blocks.extend(method_blocks)
            continue

        # ──────────────────────────────────
        # 第二步：识别独立函数
        # ──────────────────────────────────
        def_match = def_pattern.match(line)
        if def_match:
            def_indent = def_match.group(1)
            # 只处理顶层函数（不在类内部）
            if len(def_indent) == 0:
                def_start = i
                i += 1

                # 找到函数结束位置
                while i < n:
                    curr_line = lines[i]
                    if curr_line.strip() and not curr_line.startswith(' ') and not curr_line.startswith('\t'):
                        break
                    i += 1

                func_content = '\n'.join(lines[def_start:i])
                if func_content.strip():
                    blocks.append(func_content)
                continue

        # ──────────────────────────────────
        # 第三步：其他代码（直接收集）
        # ──────────────────────────────────
        other_start = i
        while i < n:
            curr_line = lines[i]
            if class_pattern.match(curr_line) or def_pattern.match(curr_line):
                break
            i += 1

        other_content = '\n'.join(lines[other_start:i])
        if other_content.strip():
            blocks.append(other_content)

    return blocks
```

**知识点**：
- ✅ **优先识别类**：先匹配 `class`，再匹配 `def`
- ✅ **缩进判断**：通过缩进来判断类的边界
- ✅ **长度阈值**：`max_class_length` 控制是否完整保留类

### index_repo 函数（重构后）

```python
def index_repo(repo_path: str, persist_dir: str = None):
    """索引代码仓库到向量数据库。"""
    Config.load()  # 加载配置

    # 从 config.yml 读取配置
    if persist_dir is None:
        persist_dir = Config.get("chroma.persist_dir", "./chroma_db")

    chunk_size = Config.get("chroma.chunk_size", 800)
    chunk_overlap = Config.get("chroma.chunk_overlap", 100)
    max_class_length = Config.get("splitting.max_class_length", 3000)
    embedding_model = Config.get("embeddings.model", "text-embedding-v3")

    all_chunks = []
    supported_exts = ('.py', '.java', '.js', '.ts', '.go', '.md')

    # 要排除的目录
    exclude_dirs = {
        'node_modules', '__pycache__', '.git', '.svn', '.hg',
        'dist', 'build', 'target', 'venv', '.venv', 'env', '.env',
        'vendor', 'bower_components', 'learn_docs'
    }

    logger.info(f"Starting repository scan: {repo_path}")

    for root, _, files in os.walk(repo_path):
        path_parts = root.replace('\\', '/').split('/')
        if any(part in exclude_dirs for part in path_parts):
            continue

        for file in files:
            if file.endswith(supported_exts):
                file_path = os.path.join(root, file)
                logger.info(f"Processing file: {file_path}")

                try:
                    loader = TextLoader(file_path, encoding='utf-8')
                    docs = loader.load()

                    # 智能代码分割
                    for doc in docs:
                        ext = os.path.splitext(file)[1]
                        # 传入 max_class_length 参数
                        blocks = split_by_code_blocks(doc.page_content, ext, max_class_length)
                        for block in blocks:
                            if block.strip():
                                splitter = RecursiveCharacterTextSplitter(
                                    chunk_size=chunk_size,
                                    chunk_overlap=chunk_overlap
                                )
                                chunks = splitter.create_documents(
                                    [block],
                                    metadatas=[{"source": file_path}]
                                )
                                all_chunks.extend(chunks)
                except Exception as e:
                    logger.warning(f"Skipping file {file_path}: {e}")

    if not all_chunks:
        logger.error("No indexable code files found")
        return

    logger.info(f"Total {len(all_chunks)} code chunks, starting vectorization...")

    # 从环境变量获取 API Key
    api_key = Config.get_env("DASHSCOPE_API_KEY")
    if not api_key:
        logger.error("DASHSCOPE_API_KEY not found in environment variables")
        return

    embeddings = DashScopeEmbeddings(
        model=embedding_model,
    )

    # 使用 langchain_chroma 的 Chroma
    vectordb = Chroma.from_documents(
        all_chunks,
        embeddings,
        persist_directory=persist_dir
    )

    logger.info(f"Indexing complete! {len(all_chunks)} chunks saved to {persist_dir}")
```

```python
@app.post("/chat")
async def chat(query: Query):
    """问答接口。"""
    vectordb = services.get("vectordb")
    llm = services.get("llm")
    summarizer_llm = services.get("summarizer_llm")
    retrieval_k = services.get("retrieval_k", 7)

    # 2. 构建原始上下文（使用 summarizer 模块）
    raw_context = build_context(docs)

    # Debug: 记录完整参考资料（仅写入日志文件）
    logger.debug("=" * 80)
    logger.debug("📚 Full reference materials (7 docs):")
    logger.debug(raw_context)
    logger.debug("=" * 80)

    # 3. 总结原始上下文（使用 summarizer 模块）
    logger.info("Summarizing retrieved documents...")
    summarized_context = summarize_context(query.question, raw_context, summarizer_llm)

    # Debug: 记录总结后的上下文
    logger.debug("=" * 80)
    logger.debug("📝 Summarized context:")
    logger.debug(summarized_context)
    logger.debug("=" * 80)

    # 4. 使用总结后的资料构建最终 Prompt
    prompt = PROMPT_TEMPLATE.format(context=summarized_context, question=query.question)

    # 5. 调用主 LLM
    response = llm.invoke(prompt)
    answer = response.content if hasattr(response, 'content') else str(response)

    # Debug: 记录模型返回结果（仅写入日志文件）
    logger.debug("=" * 80)
    logger.debug("🤖 Final model response:")
    logger.debug(answer)
    logger.debug("=" * 80)

    # 6. 记录引用来源（INFO 级别，控制台可见，显示 7 个来源）
    logger.info("=" * 60)
    logger.info("📎 References (7 docs):")
    for i, doc in enumerate(docs):
        src = doc.metadata.get("source", "unknown")
        snippet = doc.page_content[:150].replace("\n", " ")
        logger.info(f"  [{i+1}] {src}")
        logger.info(f"       Snippet: {snippet}...")
    logger.info("=" * 60)

    return {
        "answer": answer,
        "sources": [
            {
                "source": doc.metadata.get("source", "unknown"),
                "content": doc.page_content
            }
            for doc in docs
        ]
    }
```

**知识点**：
- ✅ **使用模块函数**：调用 `build_context()` 和 `summarize_context()`
- ✅ **代码简洁**：app.py 更简洁，专注于 Web 服务逻辑

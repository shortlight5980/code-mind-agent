### split_by_code_blocks 函数

```python
def split_by_code_blocks(content: str, file_ext: str):
    """按函数/类边界粗分割代码"""
    # 定义不同语言的分割模式（正则表达式）
    patterns = {
        ".py": r"\n(?=def |class )",           # Python: 匹配 def 或 class 开头的行
        ".java": r"\n(?=public |private |protected |class |interface )",
        ".js": r"\n(?=function |class |const |let |var )",
        ".ts": r"\n(?=function |class |interface |export )",
        ".go": r"\n(?=func |type |struct )",
    }
    pattern = patterns.get(file_ext)
    if pattern:
        # re.split: 按正则表达式分割字符串
        return re.split(pattern, content)
    else:
        return [content]
```

**知识点**：
- `r"\n(?=def |class )"` 是一个**正向先行断言**，意思是"匹配换行符，但这个换行符后面必须跟着 'def ' 或 'class '"
- 这样分割后，每个块都会以函数或类开头，不会把函数切两半

### index_repo 函数（核心）

```python
def index_repo(repo_path: str, persist_dir: str = "./chroma_db"):
    """索引代码仓库到向量库"""
    load_dotenv()  # 加载 .env 文件中的环境变量

    all_chunks = []
    supported_exts = ('.py', '.java', '.js', '.ts', '.go', '.md')

    # 要排除的目录列表
    exclude_dirs = {
        'node_modules', '__pycache__', '.git', '.svn', '.hg',
        'dist', 'build', 'target', 'venv', '.venv', 'env', '.env',
        'vendor', 'bower_components'
    }

    print(f"🔍 开始扫描仓库: {repo_path}")

    # os.walk: 递归遍历目录
    for root, _, files in os.walk(repo_path):
        # 检查路径中是否包含要排除的目录
        path_parts = root.replace('\\', '/').split('/')
        if any(part in exclude_dirs for part in path_parts):
            continue

        for file in files:
            if file.endswith(supported_exts):
                file_path = os.path.join(root, file)
                print(f"  处理文件: {file_path}")

                try:
                    # TextLoader: 加载文本文件
                    loader = TextLoader(file_path, encoding='utf-8')
                    docs = loader.load()

                    # 先按函数/类边界粗分
                    for doc in docs:
                        ext = os.path.splitext(file)[1]  # 获取文件扩展名
                        blocks = split_by_code_blocks(doc.page_content, ext)
                        for block in blocks:
                            if block.strip():  # 跳过空块
                                # 细粒度切分（保持上下文重叠）
                                splitter = RecursiveCharacterTextSplitter(
                                    chunk_size=800,      # 每个块最多 800 个字符
                                    chunk_overlap=100    # 块之间重叠 100 个字符
                                )
                                # create_documents: 创建文档对象
                                chunks = splitter.create_documents(
                                    [block],
                                    metadatas=[{"source": file_path}]  # 元数据：记录来源文件
                                )
                                all_chunks.extend(chunks)
                except Exception as e:
                    print(f"⚠️  跳过文件 {file_path}: {e}")
```

**知识点**：
- `chunk_size=800`：每个代码片段最多 800 字符
- `chunk_overlap=100`：相邻片段重叠 100 字符，防止上下文被切断
- `metadatas`：每个文档都可以附带元数据，这里我们存了文件路径

### 向量化入库部分

```python
    if not all_chunks:
        print("❌ 没有找到可索引的代码文件")
        return

    print(f"📦 共 {len(all_chunks)} 个代码片段，开始向量化...")

    # 初始化阿里云嵌入模型
    embeddings = DashScopeEmbeddings(
        model="text-embedding-v3",
    )

    # Chroma.from_documents: 从文档列表创建向量数据库
    vectordb = Chroma.from_documents(
        all_chunks,
        embeddings,
        persist_directory=persist_dir  # 持久化目录
    )
    vectordb.persist()  # 保存到磁盘

    print(f"✅ 索引完成！共 {len(all_chunks)} 个片段已保存到 {persist_dir}")
```

**知识点**：全局变量设为 `None`，在 `startup` 事件中初始化，这样可以更好地控制初始化时机。

### init_services 函数

```python
def init_services():
    """初始化向量库和 LLM 服务"""
    global vectordb, llm

    persist_dir = os.getenv("CHROMA_PERSIST_DIR", "./chroma_db")

    logger.info(f"正在加载向量库: {persist_dir}")

    # 初始化嵌入模型（和索引时用同一个！）
    embeddings = DashScopeEmbeddings(
        model="text-embedding-v3",
    )

    # 加载已有的向量数据库
    vectordb = Chroma(
        persist_directory=persist_dir,
        embedding_function=embeddings
    )

    # 初始化 Qwen 大模型
    llm = ChatTongyi(
        model="qwen-max",
        temperature=0.1,  # temperature 越低，回答越确定
    )

    logger.info("服务初始化完成")
```

**知识点**：
- `temperature=0.1`：控制模型的随机性。0.1 比较保守，适合问答场景
- **重要**：索引和检索必须使用**同一个嵌入模型**，否则向量空间不匹配！

### startup 事件

```python
@app.on_event("startup")
async def startup_event():
    """应用启动时初始化服务"""
```

**知识点**：`@app.on_event("startup")` 装饰的函数会在 FastAPI 应用启动时自动执行一次。

### /chat 接口（核心）

```python
@app.post("/chat")
async def chat(query: Query):
    """问答接口"""
    # 检查服务是否已初始化
    if vectordb is None or llm is None:
        return {"answer": "服务未初始化，请检查向量库是否存在"}

    logger.info(f"收到问题: {query.question}")

    # ──────────────────────────────────
    # 第一步：检索相关代码片段
    # ──────────────────────────────────
    docs = vectordb.similarity_search(query.question, k=3)
    # similarity_search: 相似度搜索，返回最相似的 k 个文档

    # ──────────────────────────────────
    # 第二步：构建 Prompt
    # ──────────────────────────────────
    context = "\n\n".join([
        f"来源: {doc.metadata.get('source', 'unknown')}\n内容:\n{doc.page_content}"
        for doc in docs
    ])

    prompt = f"""你是一个代码助手，请根据以下上下文回答问题。

上下文:
{context}

问题: {query.question}

请给出详细的回答。"""

    # ──────────────────────────────────
    # 第三步：调用 LLM 生成回答
    # ──────────────────────────────────
    response = llm.invoke(prompt)
    # 兼容处理：不同模型返回格式可能不同
    answer = response.content if hasattr(response, 'content') else str(response)

    # ──────────────────────────────────
    # 第四步：打印引用来源（调试用）
    # ──────────────────────────────────
    logger.info("=" * 60)
    logger.info("📎 引用来源:")
    for i, doc in enumerate(docs):
        src = doc.metadata.get("source", "unknown")
        snippet = doc.page_content[:150].replace("\n", " ")
        logger.info(f"  [{i+1}] {src}")
        logger.info(f"       片段: {snippet}...")
    logger.info("=" * 60)

    # ──────────────────────────────────
    # 第五步：返回结果
    # ──────────────────────────────────
    return {
        "answer": answer,
        "sources": [
            {
                "source": doc.metadata.get("source", "unknown"),
                "content": doc.page_content
            }
            for doc in docs
        ]
    }
```

## 关键设计决策

### 1. 为什么用全局变量而不是依赖注入？

为了简化代码，适合 MVP 阶段。生产环境可能会用更优雅的方式。

### 2. 为什么不使用 LangChain 的 Chain？

为了让代码更透明、更容易理解。手动实现 RAG 的每一步，初学者能看清每一步在做什么。

### 3. 为什么 chunk_size 选 800？

这是经验值：
- 太小：上下文不完整
- 太大：超过嵌入模型的处理能力，且浪费 Token

800 字符对于大多数函数来说足够了。
